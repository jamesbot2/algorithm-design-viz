/** Kahn's topological sort. Returns [status, ...order]: status 1 = acyclic (order has n nodes), 0 = cycle (order = nodes processed before it blocked). */
export function topoSort(n: number, edges: number[]): number[] {
  const adj: number[][] = Array.from({ length: n }, () => [])
  const indeg = new Array<number>(n).fill(0)
  for (let i = 0; i + 1 < edges.length; i += 2) {
    adj[edges[i]!]!.push(edges[i + 1]!) // @a: tsBuild+1
    indeg[edges[i + 1]!]!++
  }
  const queue: number[] = []
  for (let v = 0; v < n; v++) if (indeg[v] === 0) queue.push(v) // @a: tsSeed
  const order: number[] = []
  for (let head = 0; head < queue.length; head++) {
    const u = queue[head]! // @a: tsPop
    order.push(u)
    for (const v of adj[u]!) {
      indeg[v]!-- // @a: tsRelax
      if (indeg[v] === 0) queue.push(v) // @a: tsEnqueue
    }
  }
  return [order.length === n ? 1 : 0, ...order] // @a: tsResult
}
