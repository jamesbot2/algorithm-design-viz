/** DSU (union by size + path compression), teaching version. Elements are 0..n-1. */
export class DSU {
  parent: number[]
  size: number[]
  count: number // number of disjoint sets

  constructor(n: number) {
    this.parent = Array.from({ length: n }, (_, i) => i) // @a: init+2
    this.size = new Array<number>(n).fill(1)
    this.count = n
  }

  /** Root of x; afterwards every node on the path points directly at the root. */
  find(x: number): number {
    let root = x // @a: findStart
    while (this.parent[root] !== root) { // @a: findClimb+2
      root = this.parent[root]
    }
    while (this.parent[x] !== root) { // @a: compress+4
      const next = this.parent[x]
      this.parent[x] = root
      x = next
    }
    return root // @a: findReturn
  }

  /** Merge the sets of a and b. Returns true if they were different sets. */
  union(a: number, b: number): boolean {
    let ra = this.find(a), rb = this.find(b) // @a: unionFind
    if (ra === rb) return false // @a: unionSame
    if (this.size[ra] < this.size[rb]) [ra, rb] = [rb, ra] // @a: unionSwap
    this.parent[rb] = ra // @a: unionLink+2
    this.size[ra] += this.size[rb]
    this.count--
    return true // @a: unionDone
  }

  same(a: number, b: number): boolean {
    return this.find(a) === this.find(b) // @a: sameCompare
  }
}
