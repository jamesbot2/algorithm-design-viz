/** Segment tree (point assign + range sum), teaching version. Public API: 0-based, half-open [l, r). */
export class SegTree {
  n: number
  tree: number[] // tree[node] = sum of a[lo, hi); node 1 is the root, children 2*node and 2*node+1

  constructor(a: number[]) {
    this.n = a.length
    this.tree = new Array<number>(4 * Math.max(1, this.n)).fill(0)
    if (this.n > 0) this.build(a, 1, 0, this.n) // @a: buildCall
  }

  private build(a: number[], node: number, lo: number, hi: number): void {
    if (hi - lo === 1) {
      this.tree[node] = a[lo]! // @a: buildLeaf
      return
    }
    const mid = lo + ((hi - lo) >> 1)
    this.build(a, 2 * node, lo, mid)
    this.build(a, 2 * node + 1, mid, hi)
    this.tree[node] = this.tree[2 * node]! + this.tree[2 * node + 1]! // @a: buildPull
  }

  /** a[i] = v */
  set(i: number, v: number): void {
    this.setRec(1, 0, this.n, i, v) // @a: setCall
  }

  private setRec(node: number, lo: number, hi: number, i: number, v: number): void {
    if (hi - lo === 1) {
      this.tree[node] = v // @a: setLeaf
      return
    }
    const mid = lo + ((hi - lo) >> 1)
    if (i < mid) this.setRec(2 * node, lo, mid, i, v) // @a: setDescend+1
    else this.setRec(2 * node + 1, mid, hi, i, v)
    this.tree[node] = this.tree[2 * node]! + this.tree[2 * node + 1]! // @a: setPull
  }

  /** Sum of a[l..r). */
  sum(l: number, r: number): number {
    return this.query(1, 0, this.n, l, r) // @a: sumCall
  }

  private query(node: number, lo: number, hi: number, l: number, r: number): number {
    if (r <= lo || hi <= l) return 0 // @a: queryOutside
    if (l <= lo && hi <= r) return this.tree[node]! // @a: queryInside
    const mid = lo + ((hi - lo) >> 1)
    return this.query(2 * node, lo, mid, l, r) + this.query(2 * node + 1, mid, hi, l, r) // @a: querySplit
  }
}
