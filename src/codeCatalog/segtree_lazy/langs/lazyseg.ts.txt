/** Lazy segment tree (range add + range sum), teaching version. 0-based, half-open [l, r). */
export class LazySegTree {
  n: number
  total: number[] // total[node] = sum of a[lo, hi) with every tag at or above this node already applied
  lazy: number[] // lazy[node] = amount still to be added to EACH element under node (not yet pushed to children)

  constructor(a: number[]) {
    this.n = a.length
    this.total = new Array<number>(4 * Math.max(1, this.n)).fill(0)
    this.lazy = new Array<number>(4 * Math.max(1, this.n)).fill(0)
    if (this.n > 0) this.build(a, 1, 0, this.n) // @a: buildCall
  }

  private build(a: number[], node: number, lo: number, hi: number): void {
    if (hi - lo === 1) { this.total[node] = a[lo]!; return } // @a: buildLeaf
    const mid = lo + ((hi - lo) >> 1)
    this.build(a, 2 * node, lo, mid)
    this.build(a, 2 * node + 1, mid, hi)
    this.total[node] = this.total[2 * node]! + this.total[2 * node + 1]! // @a: buildPull
  }

  /** Put tag v on node: every element in [lo, hi) gains v. */
  private apply(node: number, lo: number, hi: number, v: number): void {
    this.total[node] += v * (hi - lo) // @a: applyTotal
    this.lazy[node] += v // @a: applyTag
  }

  /** Hand the pending tag of node down to both children. */
  private push(node: number, lo: number, hi: number): void {
    if (this.lazy[node] === 0) return
    const mid = lo + ((hi - lo) >> 1)
    this.apply(2 * node, lo, mid, this.lazy[node]!) // @a: pushLeft
    this.apply(2 * node + 1, mid, hi, this.lazy[node]!) // @a: pushRight
    this.lazy[node] = 0 // @a: pushClear
  }

  /** a[i] += v for every i in [l, r). */
  add(l: number, r: number, v: number): void {
    this.addRec(1, 0, this.n, l, r, v) // @a: addCall
  }

  private addRec(node: number, lo: number, hi: number, l: number, r: number, v: number): void {
    if (r <= lo || hi <= l) return // @a: addOutside
    if (l <= lo && hi <= r) { this.apply(node, lo, hi, v); return } // @a: addInside
    this.push(node, lo, hi) // @a: addPush
    const mid = lo + ((hi - lo) >> 1)
    this.addRec(2 * node, lo, mid, l, r, v) // @a: addSplit+1
    this.addRec(2 * node + 1, mid, hi, l, r, v)
    this.total[node] = this.total[2 * node]! + this.total[2 * node + 1]! // @a: addPull
  }

  /** Sum of a[l..r). */
  sum(l: number, r: number): number {
    return this.query(1, 0, this.n, l, r) // @a: sumCall
  }

  private query(node: number, lo: number, hi: number, l: number, r: number): number {
    if (r <= lo || hi <= l) return 0 // @a: queryOutside
    if (l <= lo && hi <= r) return this.total[node]! // @a: queryInside
    this.push(node, lo, hi) // @a: queryPush
    const mid = lo + ((hi - lo) >> 1)
    return this.query(2 * node, lo, mid, l, r) + this.query(2 * node + 1, mid, hi, l, r) // @a: querySplit
  }
}
