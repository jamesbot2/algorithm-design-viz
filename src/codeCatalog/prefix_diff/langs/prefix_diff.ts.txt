/** Prefix sums and difference arrays. 0-based, half-open [l, r). */
export function prefixSums(a: number[]): number[] {
  const p = new Array<number>(a.length + 1).fill(0) // @a: prefixInit
  for (let i = 0; i < a.length; i++) p[i + 1] = p[i]! + a[i]! // @a: prefixStep
  return p
}

/** Sum of a[l..r) from p = prefixSums(a). */
export function rangeSum(p: number[], l: number, r: number): number {
  return p[r]! - p[l]! // @a: rangeSumRet
}

/** Answer flat queries [l0, r0, l1, r1, ...]. */
export function rangeSums(a: number[], qs: number[]): number[] {
  const p = prefixSums(a)
  const out: number[] = []
  for (let i = 0; i + 1 < qs.length; i += 2) out.push(rangeSum(p, qs[i]!, qs[i + 1]!))
  return out
}

/** Apply flat updates [l, r, v, ...] (a[l..r) += v) with a difference array: O(n + q). */
export function rangeAddAll(a: number[], ups: number[]): number[] {
  const d = new Array<number>(a.length + 1).fill(0)
  for (let i = 0; i + 2 < ups.length; i += 3) {
    d[ups[i]!] += ups[i + 2]! // @a: diffL
    d[ups[i + 1]!] -= ups[i + 2]! // @a: diffR
  }
  const out: number[] = []
  let run = 0
  for (let i = 0; i < a.length; i++) {
    run += d[i]! // @a: restoreRun
    out.push(a[i]! + run) // @a: restoreOut
  }
  return out
}
